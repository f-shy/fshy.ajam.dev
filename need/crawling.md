```sql
SELECT rid, vorder FROM record WHERE topic='crawler-scraper' AND grp='usable' AND duplicate_of IS NULL AND coalesce(summary,'') NOT LIKE 'no free %' AND coalesce(summary,'') NOT LIKE 'no published free %' AND coalesce(summary,'') NOT LIKE 'no perpetual free %'
ORDER BY self_hosted, vorder ASC NULLS LAST
```
